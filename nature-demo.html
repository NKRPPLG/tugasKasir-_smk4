Integrate Nature 3D into Any Page

Copy and paste these 3 simple lines into your HTML or PHP application: 

<!-- 1. Include Three.js and Nature Background Scripts -->
<link rel="stylesheet" href="assets/nature-background.css">
<script src="assets/three/three.min.js"></script>
<script src="assets/nature-background.js"></script>

<!-- 2. Background container (fixed behind content) -->
<div id="nature-3d-bg"></div>

<!-- 3. Initialize with optional preset -->
<script>
  const bg = new NatureBackground3D({
    container: '#nature-3d-bg',
    preset: 'morning', // 'morning', 'noon', 'sunset', 'night'
    cinematicFlight: true,
    windSpeed: 1.0
  });
</script>
